# 0. Окружение
conda deactivate 2>/dev/null || true
source /opt/ros/lyrical/setup.bash

# 1. Сборка
colcon build --symlink-install --packages-select patrol
source install/setup.bash

# 2. Тесты (должны быть зелёные, кроме допустимого SKIPPED copyright)
python3 -m pytest src/patrol/test -v
mkdir -p evidence/pr04
python3 -m pytest src/patrol/test -v | tee evidence/pr04/tests.txt

# 3. Запуск (3 терминала, в каждом source из п.0)
# Терминал 1:
ros2 run turtlesim turtlesim_node
# Терминал 2:
ros2 run patrol patrol --ros-args -r cmd_vel:=/turtle1/cmd_vel
# Терминал 3 (диагностика):
ros2 param get /patrol linear_speed
ros2 param get /patrol turn_rate
ros2 param get /patrol publish_hz
timeout 10 ros2 topic hz /turtle1/cmd_vel --window 100

# 4. Собрать: смена частоты
ros2 param set /patrol publish_hz 5.0
ros2 param get /patrol publish_hz
timeout 10 ros2 topic hz /turtle1/cmd_vel --window 100

# 5. Собрать: отклонение недопустимого + сохранение старых speed/turn
ros2 param set /patrol publish_hz 300.0
ros2 param set /patrol linear_speed 0.25
ros2 param set /patrol turn_rate -0.5
ros2 param set /patrol linear_speed 2.0
ros2 param set /patrol turn_rate -2.0
ros2 param get /patrol linear_speed
ros2 param get /patrol turn_rate
ros2 param set /patrol publish_hz nan

# 6. Сломать (временная ветка с отключённой валидацией publish_hz)
git switch -c pr04-defect
#   в _on_set_parameters заменить validate_publish_hz(number) на number
colcon build --symlink-install --packages-select patrol && source install/setup.bash
ros2 run patrol patrol --ros-args -r cmd_vel:=/turtle1/cmd_vel   # терм. 2
ros2 param set /patrol publish_hz 0.0                            # терм. 3 -> ZeroDivisionError
git add src/patrol/patrol/patrol.py
git commit -m "PR04 defect: disable publish_hz validation"

# 7. Доказать (вернуться в основную ветку с валидацией)
git switch pr04   # или имя твоей ветки реализации
colcon build --symlink-install --packages-select patrol && source install/setup.bash
ros2 run patrol patrol --ros-args -r cmd_vel:=/turtle1/cmd_vel   # терм. 2
ros2 param set /patrol publish_hz 5.0                            # терм. 3
timeout 10 ros2 topic hz /turtle1/cmd_vel --window 100           # ~5 Hz
ros2 param set /patrol publish_hz 0.0                            # failed
ros2 param get /patrol publish_hz                                # 5.0
timeout 10 ros2 topic hz /turtle1/cmd_vel --window 100           # ~5 Hz (поток не изменился)
ros2 param set /patrol publish_hz 10.0                           # successful
timeout 10 ros2 topic hz /turtle1/cmd_vel --window 100           # ~10 Hz

# 8. Готовые сервис и action
ros2 service call /clear std_srvs/srv/Empty "{}"
ros2 action list -t

# 9. Коммиты и report.json
git add src/patrol .flake8 mypy.ini
git commit -m "PR04: runtime parameters and publish_hz validation"
IMPL_SHA=$(git rev-parse HEAD)
RECORDED_AT_UTC=$(python3 -c "from datetime import datetime, timezone; print(datetime.now(timezone.utc).strftime('%Y-%m-%dT%H:%M:%SZ'))")
cat > evidence/pr04/report.json <<JSONEOF
{
  "commit": "$IMPL_SHA",
  "recorded_at_utc": "$RECORDED_AT_UTC",
  "ai_used": true,
  "public_tests_passed": true,
  "defect_reproduced": true,
  "defect_fixed": true
}
JSONEOF
git add evidence/pr04 AI_USAGE.md
git commit -m "PR04: add parameters evidence, tests output and report"
REPORT_SHA=$(git rev-parse HEAD)

# 10. Checker
python3 .course-kit/v1/tools/check_practice.py PR04 --submission .
